                <input type="text" id="message-input" placeholder="Type your message here...">
                <button onclick="sendMessage()">Send</button>

    <script>
        const socket = new WebSocket("ws://127.0.0.1:5557");
        const chatBox = document.getElementById("chat-box");
        const msgInput = document.getElementById("message-input");

        msgInput.addEventListener("keypress", function(event) {
            if (event.key === "Enter") {
                event.preventDefault();
                sendMessage();
            }
        });

        socket.onopen = function(event) {
            chatBox.innerHTML += "<div class='message system-msg'>Successfully connected to the server!</div>";
        };

        socket.onmessage = function(event) {
            chatBox.innerHTML += "<div class='message friend-msg'><b>Friend</b>" + event.data + "</div>";
            chatBox.scrollTop = chatBox.scrollHeight; 
        };

        function sendMessage() {
            if (msgInput.value.trim() !== "") {
                socket.send(msgInput.value);
                chatBox.innerHTML += "<div class='message my-msg'><b>You</b>" + msgInput.value + "</div>";
                msgInput.value = "";
                chatBox.scrollTop = chatBox.scrollHeight; 
            }
        }
    </script>
